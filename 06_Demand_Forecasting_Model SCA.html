# create moving average baseline forecast

forecast_data = daily_demand.copy()

forecast_data["MA_7"] = (
    forecast_data["Total_Demand"]
    .shift(1)
    .rolling(window=7)
    .mean()
)

forecast_data["MA_30"] = (
    forecast_data["Total_Demand"]
    .shift(1)
    .rolling(window=30)
    .mean()
)

# evaluate moving average forecasting performance

from sklearn.metrics import mean_absolute_error, mean_squared_error

actual = test_forecast["Total_Demand"]
pred_7 = test_forecast["MA_7"]
pred_30 = test_forecast["MA_30"]

mae_7 = mean_absolute_error(actual, pred_7)
rmse_7 = np.sqrt(mean_squared_error(actual, pred_7))
mape_7 = np.mean(np.abs((actual - pred_7) / actual)) * 100

mae_30 = mean_absolute_error(actual, pred_30)
rmse_30 = np.sqrt(mean_squared_error(actual, pred_30))
mape_30 = np.mean(np.abs((actual - pred_30) / actual)) * 100

metrics_df = pd.DataFrame({
    "Model": ["7-Day Moving Average", "30-Day Moving Average"],
    "MAE": [mae_7, mae_30],
    "RMSE": [rmse_7, rmse_30],
    "MAPE (%)": [mape_7, mape_30]
})

print("Baseline Forecast Evaluation")

display(
    metrics_df.round(2)
)
